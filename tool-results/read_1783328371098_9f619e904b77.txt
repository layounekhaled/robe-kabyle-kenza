   108→  return new Date(iso).toLocaleDateString("fr-FR", {
   109→    day: "2-digit",
   110→    month: "2-digit",
   111→    year: "numeric",
   112→    hour: "2-digit",
   113→    minute: "2-digit",
   114→  });

   126→// ─── Component ───────────────────────────────────────────────────────────────
   127→
   128→export default function POSPage() {
   129→  // Product search
   130→  const [searchQuery, setSearchQuery] = useState("");
   131→  const [products, setProducts] = useState<Product[]>([]);
   132→  const [searchLoading, setSearchLoading] = useState(false);
   133→  const searchTimeout = useRef<NodeJS.Timeout | null>(null);
   134→
   135→  // Product selection
   136→  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null);
   137→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   138→  const [selectedColor, setSelectedColor] = useState<string | null>(null);
   139→  const [quantity, setQuantity] = useState(1);
   140→
   141→  // Cart
   142→  const [cart, setCart] = useState<CartItem[]>([]);
   143→  const [paymentMethod, setPaymentMethod] = useState<"cash" | "card">("cash");
   144→  const [discount, setDiscount] = useState(0);
   145→
   146→  // Checkout
   147→  const [checkoutLoading, setCheckoutLoading] = useState(false);
   148→  const [showReceipt, setShowReceipt] = useState(false);
   149→  const [lastSale, setLastSale] = useState<Sale | null>(null);
   150→
   151→  const searchInputRef = useRef<HTMLInputElement>(null);
   152→
   153→  // ─── Product Search ──────────────────────────────────────────────────────
   154→
   155→  const searchProducts = useCallback(async (query: string) => {
   156→    if (!query.trim()) {
   157→      // Load all products when search is empty
   158→      try {
   159→        const res = await fetch("/api/products?limit=50");
   160→        if (res.ok) {
   161→          const data = await res.json();
   162→          setProducts(data.products || []);
   163→        }
   164→      } catch {
   165→        // silently fail
   166→      }
   167→      return;
   168→    }
   169→
   170→    setSearchLoading(true);
   171→    try {
   172→      const res = await fetch(
   173→        `/api/products?search=${encodeURIComponent(query)}&limit=50`
   174→      );
   175→      if (res.ok) {
   176→        const data = await res.json();
   177→        setProducts(data.products || []);
   178→      }
   179→    } catch {
   180→      toast.error("Erreur lors de la recherche");
   181→    } finally {
   182→      setSearchLoading(false);
   183→    }
   184→  }, []);
   185→
   186→  // Debounced search
   187→  useEffect(() => {
   188→    if (searchTimeout.current) {
   189→      clearTimeout(searchTimeout.current);
   190→    }
   191→    searchTimeout.current = setTimeout(() => {
   192→      searchProducts(searchQuery);
   193→    }, 300);
   194→    return () => {
   195→      if (searchTimeout.current) {
   196→        clearTimeout(searchTimeout.current);
   197→      }
   198→    };
   199→  }, [searchQuery, searchProducts]);
   200→
   201→  // Initial load
   202→  useEffect(() => {
   203→    searchProducts("");
   204→  }, [searchProducts]);
   205→
   206→  // ─── Keyboard Shortcuts ──────────────────────────────────────────────────
   207→
   208→  useEffect(() => {
   209→    const handleKeyDown = (e: KeyboardEvent) => {
   210→      if (e.key === "F2") {
   211→        e.preventDefault();
   212→        searchInputRef.current?.focus();
   213→        searchInputRef.current?.select();
   214→      }
   215→      if (e.key === "F4") {
   216→        e.preventDefault();
   217→        handleCheckout();
   218→      }
   219→      if (e.key === "Escape") {
   220→        e.preventDefault();
   221→        if (showReceipt) {
   222→          handleNewSale();
   223→        } else {
   224→          handleClearCart();
   225→        }
   226→      }
   227→    };
   228→    window.addEventListener("keydown", handleKeyDown);
   229→    return () => window.removeEventListener("keydown", handleKeyDown);
   230→  }, [cart, showReceipt, paymentMethod, discount]);
   231→
   232→  // ─── Product Selection ──────────────────────────────────────────────────
   233→
   234→  const handleSelectProduct = (product: Product) => {
   235→    setSelectedProduct(product);
   236→    setSelectedSize(null);
   237→    setSelectedColor(null);
   238→    setQuantity(1);
   239→  };
   240→
   241→  const availableSizes = selectedProduct
   242→    ? [...new Set(selectedProduct.variants.map((v) => v.size))]
   243→    : [];
   244→
   245→  const availableColors = selectedProduct && selectedSize
   246→    ? [
   247→        ...new Set(
   248→          selectedProduct.variants
   249→            .filter((v) => v.size === selectedSize)
   250→            .map((v) => v.color)
   251→        ),
   252→      ]
   253→    : selectedProduct
   254→    ? [...new Set(selectedProduct.variants.map((v) => v.color))]
   255→    : [];
   256→
   257→  const selectedVariant = selectedProduct?.variants.find(
   258→    (v) => v.size === selectedSize && v.color === selectedColor
   259→  );
   260→
   261→  const variantStock = selectedVariant?.stock ?? 0;
   262→
   263→  // ─── Cart Management ────────────────────────────────────────────────────
   264→
   265→  const handleAddToCart = () => {
   266→    if (!selectedProduct || !selectedSize || !selectedColor) {
   267→      toast.error("Sélectionnez une taille et une couleur");
   268→      return;
   269→    }
   270→    if (variantStock < quantity) {
   271→      toast.error("Stock insuffisant");
   272→      return;
   273→    }
   274→    if (quantity <= 0) {
   275→      toast.error("Quantité invalide");
   276→      return;
   277→    }
   278→
   279→    const existingIndex = cart.findIndex(
   280→      (item) =>
   281→        item.productId === selectedProduct.id &&
   282→        item.size === selectedSize &&
   283→        item.color === selectedColor
   284→    );
   285→
   286→    const newQuantity =
   287→      existingIndex >= 0 ? cart[existingIndex].quantity + quantity : quantity;
   288→
   289→    // Check total quantity against stock
   290→    if (newQuantity > variantStock) {
   291→      toast.error(
   292→        `Stock insuffisant. Disponible: ${variantStock} - Dans le panier: ${existingIndex >= 0 ? cart[existingIndex].quantity : 0}`
   293→      );
   294→      return;
   295→    }
   296→
   297→    const cartItem: CartItem = {
   298→      productId: selectedProduct.id,
   299→      name: selectedProduct.name,
   300→      reference: selectedProduct.reference,
   301→      size: selectedSize,
   302→      color: selectedColor,
   303→      quantity: newQuantity,
   304→      unitPrice: selectedProduct.price,
   305→      image: selectedProduct.images[0]?.url || null,
   306→      variantId: selectedVariant!.id,
   307→    };
   308→
   309→    if (existingIndex >= 0) {
   310→      const newCart = [...cart];
   311→      newCart[existingIndex] = cartItem;
   312→      setCart(newCart);
   313→    } else {
   314→      setCart([...cart, cartItem]);
   315→    }
   316→
   317→    toast.success(`${selectedProduct.name} ajouté au panier`);
   318→    // Reset selection but keep product selected for quick re-add
   319→    setSelectedSize(null);
   320→    setSelectedColor(null);
   321→    setQuantity(1);
   322→  };
   323→
   324→  const handleUpdateCartItemQuantity = (index: number, newQty: number) => {
   325→    if (newQty <= 0) {
   326→      handleRemoveCartItem(index);
   327→      return;
   328→    }
   329→    const newCart = [...cart];
   330→    newCart[index] = { ...newCart[index], quantity: newQty };
   331→    setCart(newCart);
   332→  };
   333→
   334→  const handleRemoveCartItem = (index: number) => {
   335→    setCart(cart.filter((_, i) => i !== index));
   336→  };
   337→
   338→  const handleClearCart = () => {
   339→    if (cart.length === 0) return;
   340→    setCart([]);
   341→    setDiscount(0);
   342→    toast.info("Panier vidé");
   343→  };
   344→
   345→  // ─── Calculations ───────────────────────────────────────────────────────
   346→
   347→  const subtotal = cart.reduce(
   348→    (sum, item) => sum + item.unitPrice * item.quantity,
   349→    0
   350→  );
   351→  const total = Math.max(0, subtotal - discount);
   352→  const itemCount = cart.reduce((sum, item) => sum + item.quantity, 0);
   353→
   354→  // ─── Checkout ───────────────────────────────────────────────────────────
   355→
   356→  const handleCheckout = async () => {
   357→    if (cart.length === 0) {
   358→      toast.error("Le panier est vide");
   359→      return;
   360→    }
   361→
   362→    setCheckoutLoading(true);
   363→    try {
   364→      const res = await fetch("/api/store-sales", {
   365→        method: "POST",
   366→        headers: { "Content-Type": "application/json" },
   367→        body: JSON.stringify({
   368→          items: cart.map((item) => ({
   369→            productId: item.productId,
   370→            size: item.size,
   371→            color: item.color,
   372→            quantity: item.quantity,
   373→          })),
   374→          paymentMethod,
   375→          notes: discount > 0 ? `Remise: ${formatPrice(discount)}` : undefined,
   376→        }),
   377→      });
   378→
   379→      const data = await res.json();
   380→
   381→      if (!res.ok) {
   382→        toast.error(data.error || "Erreur lors de l'encaissement");
   383→        return;
   384→      }
   385→
   386→      setLastSale(data.sale);
   387→      setShowReceipt(true);
   388→      toast.success("Vente enregistrée avec succès !");
   389→
   390→      // Refresh products to update stock
   391→      searchProducts(searchQuery);
   392→    } catch {
   393→      toast.error("Erreur de connexion au serveur");
   394→    } finally {
   395→      setCheckoutLoading(false);
   396→    }
   397→  };
   398→
   399→  const handleNewSale = () => {
   400→    setShowReceipt(false);
   401→    setLastSale(null);
   402→    setCart([]);
   403→    setDiscount(0);
   404→    setSelectedProduct(null);
   405→    setSelectedSize(null);
   406→    setSelectedColor(null);
   407→    setQuantity(1);
   408→    searchInputRef.current?.focus();
   409→  };
   410→
   411→  const handlePrintReceipt = () => {
   412→    window.print();
   413→  };
   414→
   415→  // ─── Render ─────────────────────────────────────────────────────────────
   416→
   417→  return (
   418→    <div className="h-[calc(100vh-3.5rem)] flex flex-col lg:flex-row gap-0 overflow-hidden">
   419→      {/* ─── LEFT PANEL: Product Search & Selection (60%) ─── */}
   420→      <div className="flex-1 lg:w-[60%] flex flex-col border-r overflow-hidden">
   421→        {/* Search Bar */}
   422→        <div className="p-3 border-b bg-muted/30">
   423→          <div className="relative">
   424→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-5 h-5 text-muted-foreground" />
   425→            <Input
   426→              ref={searchInputRef}
   427→              type="text"
   428→              placeholder="Rechercher par nom, référence... (F2)"
   429→              value={searchQuery}
   430→              onChange={(e) => setSearchQuery(e.target.value)}
   431→              className="pl-10 h-12 text-base"
   432→            />
   433→            {searchQuery && (
   434→              <Button
   435→                variant="ghost"
   436→                size="sm"
   437→                className="absolute right-2 top-1/2 -translate-y-1/2 h-8 w-8 p-0"
   438→                onClick={() => {
   439→                  setSearchQuery("");
   440→                  searchInputRef.current?.focus();
   441→                }}
   442→              >
   443→                <X className="w-4 h-4" />
   444→              </Button>
   445→            )}
   446→          </div>
   447→        </div>
   448→
   449→        <div className="flex-1 flex flex-col lg:flex-row overflow-hidden">
   450→          {/* Product Grid */}
   451→          <div
   452→            className={`flex-1 overflow-hidden transition-all ${
   453→              selectedProduct ? "lg:w-1/2" : "w-full"
   454→            }`}
   455→          >
   456→            <ScrollArea className="h-full">
   457→              <div className="p-3">
   458→                {searchLoading && (
   459→                  <div className="flex items-center justify-center py-8">
   460→                    <Loader2 className="w-6 h-6 animate-spin text-kabyle-terracotta" />
   461→                  </div>
   462→                )}
   463→
   464→                {!searchLoading && products.length === 0 && (
   465→                  <div className="text-center py-12 text-muted-foreground">
   466→                    <Package className="w-12 h-12 mx-auto mb-3 opacity-50" />
   467→                    <p className="text-lg">Aucun produit trouvé</p>
   468→                    <p className="text-sm mt-1">
   469→                      Essayez un autre terme de recherche
   470→                    </p>
   471→                  </div>
   472→                )}
   473→
   474→                <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-4 gap-3">
   475→                  {products.map((product) => {
   476→                    const totalStock = product.variants.reduce(
   477→                      (sum, v) => sum + v.stock,
   478→                      0
   479→                    );
   480→                    const isSelected =
   481→                      selectedProduct?.id === product.id;
   482→
   483→                    return (
   484→                      <Card
   485→                        key={product.id}
   486→                        className={`cursor-pointer transition-all hover:shadow-md ${
   487→                          isSelected
   488→                            ? "ring-2 ring-kabyle-terracotta shadow-md"
   489→                            : "hover:ring-1 hover:ring-kabyle-terracotta/30"
   490→                        }`}
   491→                        onClick={() => handleSelectProduct(product)}
   492→                      >
   493→                        <CardContent className="p-3">
   494→                          <div className="aspect-[3/4] rounded-md overflow-hidden bg-muted mb-2 relative">
   495→                            {product.images[0] ? (
   496→                              <img
   497→                                src={product.images[0].url}
   498→                                alt={product.name}
   499→                                className="w-full h-full object-cover"
   500→                              />
   501→                            ) : (
   502→                              <div className="w-full h-full flex items-center justify-center">
   503→                                <Package className="w-8 h-8 text-muted-foreground" />
   504→                              </div>
   505→                            )}
   506→                            {totalStock === 0 && (
   507→                              <div className="absolute inset-0 bg-black/50 flex items-center justify-center">
   508→                                <Badge variant="destructive" className="text-xs">
   509→                                  Rupture
   510→                                </Badge>
   511→                              </div>
   512→                            )}
   513→                            {isSelected && (
   514→                              <div className="absolute top-1 right-1 w-6 h-6 rounded-full bg-kabyle-terracotta text-kabyle-cream flex items-center justify-center">
   515→                                <Check className="w-4 h-4" />
   516→                              </div>
   517→                            )}
   518→                          </div>
   519→                          <p className="font-semibold text-sm truncate">
   520→                            {product.name}
   521→                          </p>
   522→                          <p className="text-xs text-muted-foreground">
   523→                            {product.reference}
   524→                          </p>
   525→                          <div className="flex items-center justify-between mt-1">
   526→                            <p className="font-bold text-sm text-kabyle-terracotta">
   527→                              {formatPrice(product.price)}
   528→                            </p>
   529→                            <Badge
   530→                              variant={totalStock > 0 ? "secondary" : "destructive"}
   531→                              className="text-[10px] px-1.5 py-0"
   532→                            >
   533→                              {totalStock > 0
   534→                                ? `Stock: ${totalStock}`
   535→                                : "Rupture"}
   536→                            </Badge>
   537→                          </div>
   538→                        </CardContent>
   539→                      </Card>
   540→                    );
   541→                  })}
   542→                </div>
   543→              </div>
   544→            </ScrollArea>
   545→          </div>
   546→
   547→          {/* Variant Selector Panel */}
   548→          {selectedProduct && (
   549→            <div className="lg:w-1/2 border-t lg:border-t-0 lg:border-l overflow-y-auto bg-muted/20">
   550→              <div className="p-4">
   551→                {/* Product Info */}
   552→                <div className="flex items-start gap-3 mb-4">
   553→                  <div className="w-16 h-20 rounded-md overflow-hidden bg-muted shrink-0">
   554→                    {selectedProduct.images[0] ? (
   555→                      <img
   556→                        src={selectedProduct.images[0].url}
   557→                        alt={selectedProduct.name}
   558→                        className="w-full h-full object-cover"
   559→                      />
   560→                    ) : (
   561→                      <div className="w-full h-full flex items-center justify-center">
   562→                        <Package className="w-5 h-5 text-muted-foreground" />
   563→                      </div>
   564→                    )}
   565→                  </div>
   566→                  <div className="min-w-0">
   567→                    <h3 className="font-bold text-base">
   568→                      {selectedProduct.name}
   569→                    </h3>
   570→                    <p className="text-sm text-muted-foreground">
   571→                      Réf: {selectedProduct.reference}
   572→                    </p>
   573→                    <p className="font-bold text-lg text-kabyle-terracotta mt-1">
   574→                      {formatPrice(selectedProduct.price)}
   575→                    </p>
   576→                    {selectedProduct.fabric && (
   577→                      <p className="text-xs text-muted-foreground mt-0.5">
   578→                        Tissu: {selectedProduct.fabric}
   579→                      </p>
   580→                    )}
   581→                  </div>
   582→                </div>
   583→
   584→                <Separator className="my-4" />
   585→
   586→                {/* Size Selector */}
   587→                <div className="mb-4">
   588→                  <p className="text-sm font-semibold mb-2">Taille</p>
   589→                  <div className="flex flex-wrap gap-2">
   590→                    {availableSizes.map((size) => (
   591→                      <Button
   592→                        key={size}
   593→                        variant={selectedSize === size ? "default" : "outline"}
   594→                        className={`h-11 min-w-[3rem] text-base ${
   595→                          selectedSize === size
   596→                            ? "bg-kabyle-terracotta text-kabyle-cream hover:bg-kabyle-terracotta/90"
   597→                            : ""
   598→                        }`}
   599→                        onClick={() => {
   600→                          setSelectedSize(size);
   601→                          setSelectedColor(null);
   602→                        }}
   603→                      >
   604→                        {size}
   605→                      </Button>
   606→                    ))}
   607→                  </div>
   608→                </div>
   609→
   610→                {/* Color Selector */}
   611→                <div className="mb-4">
   612→                  <p className="text-sm font-semibold mb-2">Couleur</p>
   613→                  <div className="flex flex-wrap gap-2">
   614→                    {availableColors.map((color) => {
   615→                      const hasStock =
   616→                        selectedProduct.variants.some(
   617→                          (v) =>
   618→                            v.color === color &&
   619→                            (!selectedSize || v.size === selectedSize) &&
   620→                            v.stock > 0
   621→                        );
   622→                      return (
   623→                        <Button
   624→                          key={color}
   625→                          variant={
   626→                            selectedColor === color ? "default" : "outline"
   627→                          }
   628→                          className={`h-11 text-base gap-2 ${
   629→                            selectedColor === color
   630→                              ? "bg-kabyle-terracotta text-kabyle-cream hover:bg-kabyle-terracotta/90"
   631→                              : ""
   632→                          } ${!hasStock ? "opacity-50" : ""}`}
   633→                          disabled={!hasStock}
   634→                          onClick={() => setSelectedColor(color)}
