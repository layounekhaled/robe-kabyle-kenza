   120→function OrderFormContent() {
   121→  const searchParams = useSearchParams();
   122→  const router = useRouter();
   123→
   124→  const preselectedProductId = searchParams.get("productId");
   125→  const preselectedSize = searchParams.get("size");
   126→  const preselectedColor = searchParams.get("color");
   127→  const preselectedQty = searchParams.get("qty");
   128→
   129→  // Product selection
   130→  const [products, setProducts] = useState<Product[]>([]);
   131→  const [selectedProductId, setSelectedProductId] = useState<string | null>(
   132→    preselectedProductId
   133→  );
   134→  const [selectedSize, setSelectedSize] = useState<string | null>(
   135→    preselectedSize
   136→  );
   137→  const [selectedColor, setSelectedColor] = useState<string | null>(
   138→    preselectedColor
   139→  );
   140→  const [quantity, setQuantity] = useState(
   141→    preselectedQty ? parseInt(preselectedQty) : 1
   142→  );
   143→  const [loadingProducts, setLoadingProducts] = useState(true);
   144→  const [searchQuery, setSearchQuery] = useState("");
   145→
   146→  // Customer info
   147→  const [wilayas, setWilayas] = useState<Wilaya[]>([]);
   148→  const [communes, setCommunes] = useState<Commune[]>([]);
   149→  const [loadingWilayas, setLoadingWilayas] = useState(false);
   150→  const [loadingCommunes, setLoadingCommunes] = useState(false);
   151→
   152→  // Shipping
   153→  const [deliveryType, setDeliveryType] = useState<"home" | "stopdesk">("home");
   154→  const [shippingRates, setShippingRates] = useState<{ home: number; stopDesk: number; source: string } | null>(null);
   155→  const [shippingCost, setShippingCost] = useState<number | null>(null);
   156→  const [loadingShipping, setLoadingShipping] = useState(false);

   158→
   159→  // Order result
   160→  const [orderResult, setOrderResult] = useState<{
   161→    orderNumber: string;
   162→    totalAmount: number;
   163→  } | null>(null);
   164→  const [submitted, setSubmitted] = useState(false);
   165→  const [submitting, setSubmitting] = useState(false);
   166→
   167→  // Form
   168→  const form = useForm<CustomerFormValues>({
   169→    resolver: zodResolver(customerSchema),
   170→    defaultValues: {
   171→      name: "",
   172→      phone: "",
   173→      wilayaId: "",
   174→      communeId: "",
   175→      address: "",
   176→      notes: "",
   177→    },
   178→  });
   179→
   180→  // Selected product details
   181→  const selectedProduct = products.find((p) => p.id === selectedProductId);
   182→  const sizes = selectedProduct
   183→    ? [...new Set(selectedProduct.variants.map((v) => v.size))]
   184→    : [];
   185→  const colorsForSize = selectedProduct && selectedSize
   186→    ? [...new Set(selectedProduct.variants.filter((v) => v.size === selectedSize).map((v) => v.color))]
   187→    : selectedProduct
   188→    ? [...new Set(selectedProduct.variants.map((v) => v.color))]
   189→    : [];
   190→  const inStockColorsForSize = selectedSize
   191→    ? selectedProduct?.variants
   192→        .filter((v) => v.size === selectedSize && v.stock > 0)
   193→        .map((v) => v.color) || []
   194→    : colorsForSize;
   195→  const sizesForColor = selectedColor
   196→    ? [...new Set(selectedProduct?.variants.filter((v) => v.color === selectedColor).map((v) => v.size))] || []
   197→    : sizes;
   198→  const inStockSizesForColor = selectedColor
   199→    ? selectedProduct?.variants
   200→        .filter((v) => v.color === selectedColor && v.stock > 0)
   201→        .map((v) => v.size) || []
   202→    : sizes;
   203→
   204→  const selectedVariant = selectedProduct?.variants.find(
   205→    (v) => v.size === selectedSize && v.color === selectedColor
   206→  );
   207→  const currentStock = selectedVariant?.stock ?? 0;
   208→
   209→  // Filtered products for search
   210→  const filteredProducts = products.filter(
   211→    (p) =>
   212→      !searchQuery ||
   213→      p.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   214→      p.reference.toLowerCase().includes(searchQuery.toLowerCase())
   215→  );
   216→
   217→  // Load products

   219→    async function loadProducts() {
   220→      try {
   221→        const res = await fetch("/api/products?limit=100");
   222→        if (res.ok) {
   223→          const data = await res.json();
   224→          setProducts(data.products || []);
   225→        }
   226→      } catch {
   227→        // silently fail
   228→      } finally {
   229→        setLoadingProducts(false);
   230→      }
   231→    }
   232→    loadProducts();
   233→  }, []);
   234→
   235→  // Load wilayas and shipping fees
   236→  useEffect(() => {
   237→    async function loadWilayas() {
   238→      setLoadingWilayas(true);
   239→      try {
   240→        const res = await fetch("/api/ecotrack?action=wilayas");
   241→        if (res.ok) {
   242→          const data = await res.json();
   243→          setWilayas(data.wilayas || []);
   244→        }
   245→      } catch {
   246→        toast.error("Impossible de charger les wilayas. Veuillez réessayer.");
   247→      } finally {
   248→        setLoadingWilayas(false);
   249→      }
   250→    }

   252→      try {
   253→        const res = await fetch("/api/ecotrack?action=fees");
   254→        if (res.ok) {
   255→          const data = await res.json();
   256→          setAllFees(data.fees || null);
   257→        }
   258→      } catch {
   259→        // Fees will be fetched per wilaya as fallback
   260→      }
   261→    }
   262→    loadWilayas();
   263→    loadFees();
   264→  }, []);
   265→
   266→  // Load communes when wilaya changes

   274→      setLoadingCommunes(true);

   276→        const res = await fetch(
   277→          `/api/ecotrack?action=communes&wilayaId=${selectedWilayaId}`

   280→          const data = await res.json();
   281→          setCommunes(data.communes || []);
   282→        }
   283→      } catch {
   284→        toast.error("Impossible de charger les communes");
   285→      } finally {
   286→        setLoadingCommunes(false);
   287→      }
   288→    }
   289→    loadCommunes();
   290→    form.setValue("communeId", "");
   291→  }, [selectedWilayaId, form]);
   292→
   293→  // Reset commune when delivery type changes (stop desk requires specific communes)
   294→  useEffect(() => {
   295→    form.setValue("communeId", "");
   296→  }, [deliveryType, form]);
   297→
   298→  // Filtered communes based on delivery type
   299→  const filteredCommunes = deliveryType === "stopdesk"
   300→    ? communes.filter((c) => c.hasStopDesk)
   301→    : communes;
   302→
   303→  // Check if stop desk is available in the selected wilaya
   304→  const hasStopDeskInWilaya = communes.some((c) => c.hasStopDesk);
   305→
   306→  // Calculate shipping rates when wilaya is selected
   307→  useEffect(() => {
   308→    if (!selectedWilayaId) {
   309→      setShippingCost(null);
   310→      setShippingRates(null);
   311→      return;
   312→    }
   313→    async function calcShipping() {
   314→      setLoadingShipping(true);
   315→      try {
   316→        const wilayaCode = parseInt(selectedWilayaId);
   317→        // First try to use cached fees
   318→        if (allFees && allFees[wilayaCode]) {
   319→          const rates = allFees[wilayaCode];
   320→          setShippingRates({ home: rates.home, stopDesk: rates.stopDesk, source: "api" });
   321→          setShippingCost(deliveryType === "home" ? rates.home : rates.stopDesk);
   322→          setLoadingShipping(false);
   323→          return;
   324→        }
   325→        // Fallback: fetch rates for this specific wilaya
   326→        const res = await fetch(
   327→          `/api/ecotrack?action=rates&wilayaId=${selectedWilayaId}`
   328→        );
   329→        if (res.ok) {
   330→          const data = await res.json();
   331→          const rates = data.rates;
   332→          setShippingRates(rates);
   333→          // Set shipping cost based on current delivery type
   334→          setShippingCost(deliveryType === "home" ? rates.home : rates.stopDesk);
   335→        }
   336→      } catch {
   337→        setShippingCost(null);
   338→        setShippingRates(null);
   339→      } finally {
   340→        setLoadingShipping(false);
   341→      }
   342→    }
   343→    calcShipping();
   344→  }, [selectedWilayaId, deliveryType, allFees]);
   345→
   346→  // Submit handler
   347→  const handleSubmitOrder = async () => {
   348→    if (!selectedProduct || !selectedSize || !selectedColor) {
   349→      toast.error("Veuillez sélectionner un produit, une taille et une couleur");
   350→      return;
   351→    }
   352→    if (currentStock === 0) {
   353→      toast.error("Ce produit est en rupture de stock");
   354→      return;
   355→    }
   356→
   357→    const formValid = await form.trigger();
   358→    if (!formValid) return;
   359→
   360→    setSubmitting(true);
   361→    try {
   362→      const formValues = form.getValues();
   363→      // wilayaId is the wilaya code (e.g., "16" for Alger)
   364→      const selectedWilaya = wilayas.find(
   365→        (w) => String(w.code) === formValues.wilayaId
   366→      );
   367→      // communeId is the commune name (e.g., "Alger Centre")
   368→      const selectedCommune = communes.find(
   369→        (c) => c.name === formValues.communeId
   370→      );
   371→
   372→      const payload = {
   373→        customerName: formValues.name,
   374→        customerPhone: formValues.phone,
   375→        customerWilaya: selectedWilaya?.name || formValues.wilayaId,
   376→        customerWilayaCode: formValues.wilayaId,
   377→        customerCommune: selectedCommune?.name || formValues.communeId,
   378→        customerAddress: formValues.address,
   379→        items: [
   380→          {
   381→            productId: selectedProduct.id,
   382→            size: selectedSize,
   383→            color: selectedColor,
   384→            quantity,
   385→          },
   386→        ],
   387→        notes: formValues.notes || undefined,
   388→        shippingCost: shippingCost || 0,
   389→        deliveryType,
   390→      };
   391→
   392→      const res = await fetch("/api/orders", {
   393→        method: "POST",
   394→        headers: { "Content-Type": "application/json" },
   395→        body: JSON.stringify(payload),
   396→      });
   397→
   398→      if (res.ok) {
   399→        const data = await res.json();
   400→        setOrderResult({
   401→          orderNumber: data.order.orderNumber,
   402→          totalAmount: data.order.totalAmount,
   403→        });
   404→        setSubmitted(true);
   405→        toast.success("Commande créée avec succès !");
   406→      } else {
   407→        const error = await res.json();
   408→        toast.error(error.error || "Erreur lors de la commande");
   409→      }
   410→    } catch {
   411→      toast.error("Erreur de connexion. Veuillez réessayer.");
   412→    } finally {
   413→      setSubmitting(false);
   414→    }
   415→  };
   416→
   417→  const itemTotal = selectedProduct ? selectedProduct.price * quantity : 0;
   418→  const totalWithShipping = itemTotal + (shippingCost || 0);
   419→
   420→  // ─────────── Confirmation screen ───────────
   421→  if (submitted && orderResult) {
   422→    return (
   423→      <div className="min-h-screen flex flex-col">
   424→        <Navbar />
   425→        <main className="flex-1 flex items-center justify-center px-4">
   426→          <div className="text-center space-y-6 max-w-md w-full">
   427→            <div className="mx-auto flex h-20 w-20 items-center justify-center rounded-full bg-green-100">
   428→              <PartyPopper className="h-10 w-10 text-green-600" />
   429→            </div>
   430→            <h2 className="text-2xl font-bold text-kabyle-dark">
   431→              Commande Confirmée !
   432→            </h2>
   433→            <p className="text-muted-foreground">
   434→              Votre commande a été enregistrée avec succès. Vous recevrez une
   435→              confirmation par téléphone.
   436→            </p>
   437→            <Card className="border-kabyle-terracotta/20">
   438→              <CardContent className="p-6 space-y-3">

   440→                  <p className="text-sm text-muted-foreground">Numéro de commande</p>

   442→                    {orderResult.orderNumber}
   443→                  </p>
   444→                </div>
   445→                <Separator />
   446→                <div>
   447→                  <p className="text-sm text-muted-foreground">Montant total</p>

   449→                    {formatPrice(orderResult.totalAmount + (shippingCost || 0))}
   450→                  </p>
   451→                </div>
   452→              </CardContent>
   453→            </Card>
   454→            <div className="flex flex-col sm:flex-row gap-3 justify-center">
   455→              <Link href="/catalog">
   456→                <Button variant="outline" className="border-kabyle-terracotta text-kabyle-terracotta">
   457→                  <ArrowLeft className="mr-2 h-4 w-4" />
   458→                  Continuer les achats
   459→                </Button>
   460→              </Link>
   461→              <Link href="/">
   462→                <Button className="bg-kabyle-terracotta hover:bg-kabyle-red text-white">
   463→                  Retour à l&apos;accueil
   464→                </Button>
   465→              </Link>
   466→            </div>
   467→          </div>
   468→        </main>
   469→        <Footer />
   470→      </div>
   471→    );
   472→  }
   473→
   474→  // ─────────── Main unified form ───────────
   475→  return (
   476→    <div className="min-h-screen flex flex-col">
   477→      <Navbar />
   478→
   479→      <main className="flex-1">
   480→        {/* Header */}
   481→        <div className="bg-kabyle-cream/50 kabyle-pattern border-b">
   482→          <div className="mx-auto max-w-6xl px-4 sm:px-6 lg:px-8 py-6">
   483→            <h1 className="text-2xl sm:text-3xl font-bold text-kabyle-dark">
   484→              <ShoppingBag className="inline-block mr-2 h-7 w-7 text-kabyle-terracotta" />
   485→              Commander
   486→            </h1>
   487→            <p className="mt-1 text-muted-foreground">
   488→              Remplissez le formulaire ci-dessous pour passer votre commande
   489→            </p>
   490→          </div>
   491→        </div>
   492→
   493→        <div className="mx-auto max-w-6xl px-4 sm:px-6 lg:px-8 py-6">
   494→          <Form {...form}>
   495→            <form onSubmit={form.handleSubmit(handleSubmitOrder)}>
   496→              <div className="grid lg:grid-cols-5 gap-6 lg:gap-8">
   497→
   498→                {/* ═══════ LEFT: Product Selection (3 cols) ═══════ */}
   499→                <div className="lg:col-span-3 space-y-6">
   500→
   501→                  {/* ── Section: Produit ── */}
   502→                  <Card>
   503→                    <CardContent className="p-5 space-y-5">
   504→                      <h2 className="text-lg font-semibold text-kabyle-dark flex items-center gap-2">
   505→                        <ShoppingBag className="h-5 w-5 text-kabyle-terracotta" />
   506→                        Produit
   507→                      </h2>
   508→
   509→                      {!selectedProductId ? (
   510→                        <>
   511→                          {/* Search */}
   512→                          <div className="relative">
   513→                            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   514→                            <Input
   515→                              placeholder="Rechercher par nom ou référence..."
   516→                              value={searchQuery}
   517→                              onChange={(e) => setSearchQuery(e.target.value)}
   518→                              className="pl-10"
   519→                            />
   520→                          </div>
   521→
   522→                          {loadingProducts ? (
   523→                            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   524→                              {Array.from({ length: 6 }).map((_, i) => (
   525→                                <div key={i} className="animate-pulse rounded-lg border bg-muted h-44" />
   526→                              ))}
   527→                            </div>
   528→                          ) : (
   529→                            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3 max-h-[55vh] overflow-y-auto custom-scrollbar pr-1">
   530→                              {filteredProducts.map((product) => (
   531→                                <button
   532→                                  key={product.id}
   533→                                  type="button"
   534→                                  onClick={() => {
   535→                                    setSelectedProductId(product.id);
   536→                                    setSelectedSize(null);
   537→                                    setSelectedColor(null);
   538→                                    setQuantity(1);
   539→                                  }}
   540→                                  className="group text-left rounded-lg border overflow-hidden hover:border-kabyle-terracotta hover:shadow-md transition-all"
   541→                                >
   542→                                  <div className="relative aspect-[3/4] bg-muted">
   543→                                    <Image
   544→                                      src={product.images.length > 0 ? product.images[0].url : "/logo.svg"}
   545→                                      alt={product.name}
   546→                                      fill
   547→                                      sizes="33vw"
   548→                                      className="object-cover group-hover:scale-105 transition-transform duration-300"
   549→                                    />
   550→                                  </div>
   551→                                  <div className="p-2">
   552→                                    <p className="text-xs text-muted-foreground font-mono">{product.reference}</p>
   553→                                    <p className="text-sm font-medium text-kabyle-dark line-clamp-1">{product.name}</p>
   554→                                    <p className="text-sm font-bold text-kabyle-terracotta">{formatPrice(product.price)}</p>
   555→                                  </div>
   556→                                </button>
   557→                              ))}
   558→                              {filteredProducts.length === 0 && (
   559→                                <p className="col-span-full text-center text-muted-foreground py-8">
   560→                                  Aucun produit trouvé
   561→                                </p>
   562→                              )}
   563→                            </div>
   564→                          )}
   565→                        </>
   566→                      ) : (
   567→                        <div className="space-y-5">
   568→                          {/* Selected product banner */}
   569→                          <div className="flex gap-4 p-4 rounded-xl border bg-kabyle-cream/20">
   570→                            <div className="relative h-24 w-20 shrink-0 rounded-lg overflow-hidden bg-muted">
   571→                              <Image
   572→                                src={selectedProduct?.images?.[0]?.url || "/logo.svg"}
   573→                                alt={selectedProduct?.name || ""}
   574→                                fill
   575→                                sizes="80px"
   576→                                className="object-cover"
   577→                              />
   578→                            </div>
