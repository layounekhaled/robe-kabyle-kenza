    91→    91→export default function ProductsPage() {
    92→    92→  const router = useRouter();
    93→    93→  const [products, setProducts] = useState<Product[]>([]);
    94→    94→  const [pagination, setPagination] = useState<Pagination>({ page: 1, limit: 20, total: 0, totalPages: 0 });
    95→    95→  const [loading, setLoading] = useState(true);
    96→    96→  const [search, setSearch] = useState('');
    97→    97→
    98→    98→  // Selection state
    99→    99→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   100→   100→
   101→   101→  // Dialog states
   102→   102→  const [dialogOpen, setDialogOpen] = useState(false);
   103→   103→  const [editProduct, setEditProduct] = useState<Product | null>(null);
   104→   104→  const [saving, setSaving] = useState(false);
   105→   105→
   106→   106→  // Delete state (single)
   107→   107→  const [deleteProduct, setDeleteProduct] = useState<Product | null>(null);
   108→   108→  const [deleting, setDeleting] = useState(false);
   109→   109→
   110→   110→  // Bulk delete state
   111→   111→  const [bulkDeleteOpen, setBulkDeleteOpen] = useState(false);
   112→   112→  const [bulkDeleting, setBulkDeleting] = useState(false);
   113→   113→
   114→   114→  // Form state
   115→   115→  const [formReference, setFormReference] = useState('');
   116→   116→  const [formName, setFormName] = useState('');
   117→   117→  const [formDescription, setFormDescription] = useState('');
   118→   118→  const [formPrice, setFormPrice] = useState('');
   119→   119→  const [formFabric, setFormFabric] = useState('');
   120→   120→  const [formFeatured, setFormFeatured] = useState(false);
   121→   121→  const [formActive, setFormActive] = useState(true);
   122→   122→  const [formVariants, setFormVariants] = useState<Array<{ size: string; color: string; stock: string }>>([
   123→   123→    { size: '', color: '', stock: '0' },
   124→   124→  ]);
   125→   125→  const [formImages, setFormImages] = useState<ImageItem[]>([]);
   126→   126→
   127→   127→  const fetchProducts = useCallback(async (page = 1) => {
   128→   128→    try {
   129→   129→      setLoading(true);
   130→   130→      const params = new URLSearchParams();

   132→   132→      params.set('page', page.toString());
   133→   133→      params.set('limit', '20');
   134→   134→
   135→   135→      const res = await fetch(`/api/products?${params}`);
   136→   136→      if (res.ok) {
   137→   137→        const data = await res.json();
   138→   138→        setProducts(data.products);
   139→   139→        setPagination(data.pagination);
   140→   140→        setSelectedIds(new Set()); // Clear selection on new data
   141→   141→      }
   142→   142→    } catch (error) {
   143→   143→      console.error('Error fetching products:', error);
   144→   144→    } finally {
   145→   145→      setLoading(false);
   146→   146→    }
   147→   147→  }, [search]);
   148→   148→
   149→   149→  useEffect(() => {
   150→   150→    fetchProducts(1);
   151→   151→  }, [fetchProducts]);
   152→   152→
   153→   153→  // Selection helpers
   154→   154→  const toggleSelect = (id: string) => {
   155→   155→    setSelectedIds((prev) => {
   156→   156→      const next = new Set(prev);
   157→   157→      if (next.has(id)) {
   158→   158→        next.delete(id);
   159→   159→      } else {
   160→   160→        next.add(id);
   161→   161→      }
   162→   162→      return next;
   163→   163→    });
   164→   164→  };
   165→   165→
   166→   166→  const toggleSelectAll = () => {
   167→   167→    if (selectedIds.size === products.length) {
   168→   168→      setSelectedIds(new Set());
   169→   169→    } else {
   170→   170→      setSelectedIds(new Set(products.map((p) => p.id)));
   171→   171→    }
   172→   172→  };
   173→   173→
   174→   174→  const allSelected = products.length > 0 && selectedIds.size === products.length;
   175→   175→  const someSelected = selectedIds.size > 0 && !allSelected;
   176→   176→
   177→   177→  const openCreateDialog = () => {
   178→   178→    setEditProduct(null);
   179→   179→    setFormReference('');
   180→   180→    setFormName('');
   181→   181→    setFormDescription('');
   182→   182→    setFormPrice('');
   183→   183→    setFormFabric('');
   184→   184→    setFormFeatured(false);
   185→   185→    setFormActive(true);
   186→   186→    setFormVariants([{ size: '', color: '', stock: '0' }]);
   187→   187→    setFormImages([]);
   188→   188→    setDialogOpen(true);
   189→   189→  };
   190→   190→
   191→   191→  const openEditDialog = (product: Product) => {
   192→   192→    setEditProduct(product);
   193→   193→    setFormReference(product.reference);
   194→   194→    setFormName(product.name);
   195→   195→    setFormDescription(product.description || '');
   196→   196→    setFormPrice(product.price.toString());
   197→   197→    setFormFabric(product.fabric || '');
   198→   198→    setFormFeatured(product.featured);
   199→   199→    setFormActive(product.active);
   200→   200→    setFormVariants(
   201→   201→      product.variants.length > 0
   202→   202→        ? product.variants.map((v) => ({ size: v.size, color: v.color, stock: v.stock.toString() }))
   203→   203→        : [{ size: '', color: '', stock: '0' }]
   204→   204→    );
   205→   205→    const imageItems: ImageItem[] = (product.images || []).map((img, index) => ({
   206→   206→      id: `existing-${img.id}`,
   207→   207→      url: img.url,
   208→   208→      alt: img.alt || undefined,
   209→   209→      sortOrder: img.sortOrder ?? index,
   210→   210→    }));
   211→   211→    setFormImages(imageItems);
   212→   212→    setDialogOpen(true);
   213→   213→  };
   214→   214→
   215→   215→  const handleSave = async () => {
   216→   216→    if (!formReference || !formName || !formPrice) {
   217→   217→      toast.error('Veuillez remplir la référence, le nom et le prix');
   218→   218→      return;
   219→   219→    }
   220→   220→
   221→   221→    // Check if any images are still uploading
   222→   222→    const uploadingImages = formImages.filter((img) => img.isUploading);
   223→   223→    if (uploadingImages.length > 0) {
   224→   224→      toast.error('Veuillez attendre la fin des uploads en cours');
   225→   225→      return;
   226→   226→    }
   227→   227→
   228→   228→    // Check for upload errors
   229→   229→    const errorImages = formImages.filter((img) => img.uploadError);
   230→   230→    if (errorImages.length > 0) {
   231→   231→      toast.error('Certaines images ont échoué. Supprimez-les ou réessayez.');
   232→   232→      return;
   233→   233→    }
   234→   234→
   235→   235→    const validVariants = formVariants.filter((v) => v.size && v.color);
   236→   236→
   237→   237→    const imagesData = formImages.map((img, index) => ({
   238→   238→      url: img.url,
   239→   239→      alt: img.alt || null,
   240→   240→      sortOrder: index,
   241→   241→    }));
   242→   242→
   243→   243→    const payload = {
   244→   244→      reference: formReference,
   245→   245→      name: formName,
   246→   246→      description: formDescription || null,
   247→   247→      price: parseFloat(formPrice),
   248→   248→      fabric: formFabric || null,
   249→   249→      featured: formFeatured,
   250→   250→      active: formActive,
   251→   251→      variants: validVariants.map((v) => ({ size: v.size, color: v.color, stock: parseInt(v.stock) || 0 })),
   252→   252→      images: imagesData,
   253→   253→    };

   255→   255→    try {
   256→   256→      setSaving(true);
   257→   257→      let res;
   258→   258→      if (editProduct) {
   259→   259→        res = await fetch(`/api/products/${editProduct.id}`, {
   260→   260→          method: 'PUT',
   261→   261→          headers: { 'Content-Type': 'application/json' },
   262→   262→          body: JSON.stringify(payload),
   263→   263→        });
   264→   264→      } else {
   265→   265→        res = await fetch('/api/products', {
   266→   266→          method: 'POST',
   267→   267→          headers: { 'Content-Type': 'application/json' },
   268→   268→          body: JSON.stringify(payload),
   269→   269→        });
   270→   270→      }
   271→   271→
   272→   272→      if (res.ok) {
   273→   273→        toast.success(editProduct ? 'Produit mis à jour avec succès' : 'Produit créé avec succès');
   274→   274→        setDialogOpen(false);
   275→   275→        fetchProducts(pagination.page);
   276→   276→      } else {
   277→   277→        const data = await res.json();
   278→   278→        toast.error(data.error || 'Erreur lors de la sauvegarde');
   279→   279→      }
   280→   280→    } catch {
   281→   281→      toast.error('Erreur lors de la sauvegarde');
   282→   282→    } finally {
   283→   283→      setSaving(false);
   284→   284→    }
   285→   285→  };
   286→   286→
   287→   287→  const handleDelete = async () => {
   288→   288→    if (!deleteProduct) return;
   289→   289→
   290→   290→    try {
   291→   291→      setDeleting(true);
   292→   292→      const res = await fetch(`/api/products/${deleteProduct.id}`, { method: 'DELETE' });
   293→   293→      if (res.ok) {
   294→   294→        toast.success('Produit supprimé avec succès');
   295→   295→        setDeleteProduct(null);
   296→   296→        fetchProducts(pagination.page);
   297→   297→      } else {
   298→   298→        const data = await res.json();
   299→   299→        toast.error(data.error || 'Erreur lors de la suppression');
   300→   300→      }
   301→   301→    } catch {
   302→   302→      toast.error('Erreur lors de la suppression');
   303→   303→    } finally {
   304→   304→      setDeleting(false);
   305→   305→    }
   306→   306→  };
   307→   307→
   308→   308→  const handleBulkDelete = async () => {
   309→   309→    if (selectedIds.size === 0) return;
   310→   310→
   311→   311→    try {
   312→   312→      setBulkDeleting(true);
   313→   313→      const res = await fetch('/api/products', {
   314→   314→        method: 'DELETE',
   315→   315→        headers: { 'Content-Type': 'application/json' },
   316→   316→        body: JSON.stringify({ ids: Array.from(selectedIds) }),
   317→   317→      });
   318→   318→      if (res.ok) {
   319→   319→        const data = await res.json();
   320→   320→        toast.success(data.message || `${selectedIds.size} produits supprimés avec succès`);
   321→   321→        setSelectedIds(new Set());
   322→   322→        setBulkDeleteOpen(false);
   323→   323→        fetchProducts(pagination.page);
   324→   324→      } else {
   325→   325→        const data = await res.json();
   326→   326→        toast.error(data.error || 'Erreur lors de la suppression');
   327→   327→      }
   328→   328→    } catch {
   329→   329→      toast.error('Erreur lors de la suppression multiple');
   330→   330→    } finally {
   331→   331→      setBulkDeleting(false);
   332→   332→    }
   333→   333→  };
   334→   334→
   335→   335→  const addVariant = () => {
   336→   336→    setFormVariants((prev) => [...prev, { size: '', color: '', stock: '0' }]);
   337→   337→  };
   338→   338→
   339→   339→  const removeVariant = (index: number) => {
   340→   340→    setFormVariants((prev) => prev.filter((_, i) => i !== index));
   341→   341→  };
   342→   342→
   343→   343→  const updateVariant = (index: number, field: 'size' | 'color' | 'stock', value: string) => {
   344→   344→    setFormVariants((prev) =>
   345→   345→      prev.map((v, i) => (i === index ? { ...v, [field]: value } : v))
   346→   346→    );
   347→   347→  };
   348→   348→
   349→   349→  const getTotalStock = (product: Product) => {
   350→   350→    return product.variants.reduce((sum, v) => sum + v.stock, 0);
   351→   351→  };
   352→   352→
   353→   353→  return (
   354→   354→    <div className="space-y-6">
   355→   355→      {/* Header */}
   356→   356→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   357→   357→        <div>
   358→   358→          <h2 className="text-2xl font-bold">Produits</h2>
   359→   359→          <p className="text-muted-foreground">{pagination.total} produits au total</p>
   360→   360→        </div>
   361→   361→        <div className="flex items-center gap-2">
   362→   362→          {selectedIds.size > 0 && (
   363→   363→            <Button
   364→   364→              variant="destructive"
   365→   365→              onClick={() => setBulkDeleteOpen(true)}
   366→   366→              className="shrink-0"
   367→   367→            >
   368→   368→              <Trash2 className="mr-2 h-4 w-4" />
   369→   369→              Supprimer ({selectedIds.size})
   370→   370→            </Button>
   371→   371→          )}
   372→   372→          <Button onClick={openCreateDialog} className="bg-primary hover:bg-primary/90">
   373→   373→            <Plus className="mr-2 h-4 w-4" />
   374→   374→            Ajouter un produit
   375→   375→          </Button>
   376→   376→        </div>
   377→   377→      </div>
   378→   378→
   379→   379→      {/* Search */}
   380→   380→      <div className="relative max-w-md">
   381→   381→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   382→   382→        <Input
   383→   383→          placeholder="Rechercher par référence, nom..."
   384→   384→          value={search}
   385→   385→          onChange={(e) => setSearch(e.target.value)}
   386→   386→          className="pl-10"
   387→   387→        />
   388→   388→      </div>
   389→   389→
   390→   390→      {/* Selection info bar */}
   391→   391→      {selectedIds.size > 0 && (
   392→   392→        <div className="flex items-center gap-3 px-4 py-2 bg-muted/50 rounded-lg border">
   393→   393→          <span className="text-sm font-medium">
   394→   394→            {selectedIds.size} produit{selectedIds.size > 1 ? 's' : ''} sélectionné{selectedIds.size > 1 ? 's' : ''}
   395→   395→          </span>
   396→   396→          <Button
   397→   397→            variant="ghost"
   398→   398→            size="sm"
   399→   399→            className="text-xs h-7"
   400→   400→            onClick={() => setSelectedIds(new Set())}
   401→   401→          >
   402→   402→            Tout désélectionner
   403→   403→          </Button>
   404→   404→        </div>
   405→   405→      )}
   406→   406→
   407→   407→      {/* Products Table */}
   408→   408→      <Card>
   409→   409→        <CardContent className="p-0">
   410→   410→          {loading ? (
   411→   411→            <div className="p-6 space-y-4">
   412→   412→              {Array.from({ length: 5 }).map((_, i) => (
   413→   413→                <Skeleton key={i} className="h-16 w-full" />
   414→   414→              ))}
   415→   415→            </div>
   416→   416→          ) : products.length === 0 ? (
   417→   417→            <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   418→   418→              <ImageIcon className="h-12 w-12 mb-4 opacity-50" />
   419→   419→              <p className="text-lg font-medium">Aucun produit trouvé</p>
   420→   420→              <p className="text-sm">Commencez par ajouter votre premier produit</p>
   421→   421→            </div>
   422→   422→          ) : (
   423→   423→            <>
   424→   424→              <div className="overflow-x-auto">
   425→   425→                <Table>
   426→   426→                  <TableHeader>
   427→   427→                    <TableRow>
   428→   428→                      <TableHead className="w-12">
   429→   429→                        <Checkbox
   430→   430→                          checked={allSelected}
   431→   431→                          ref={(el) => {
   432→   432→                            if (el) {
   433→   433→                              (el as HTMLButtonElement & { indeterminate?: boolean }).indeterminate = someSelected;
   434→   434→                            }
   435→   435→                          }}
   436→   436→                          onCheckedChange={toggleSelectAll}
   437→   437→                        />
   438→   438→                      </TableHead>
   439→   439→                      <TableHead className="w-16">Image</TableHead>
   440→   440→                      <TableHead>Référence</TableHead>
   441→   441→                      <TableHead>Nom</TableHead>
   442→   442→                      <TableHead>Prix</TableHead>
   443→   443→                      <TableHead>Stock total</TableHead>
   444→   444→                      <TableHead>Statut</TableHead>
   445→   445→                      <TableHead className="text-right">Actions</TableHead>
   446→   446→                    </TableRow>
   447→   447→                  </TableHeader>
   448→   448→                  <TableBody>
   449→   449→                    {products.map((product) => (
   450→   450→                      <TableRow
   451→   451→                        key={product.id}
   452→   452→                        className={selectedIds.has(product.id) ? 'bg-muted/50' : ''}
   453→   453→                      >
   454→   454→                        <TableCell>
   455→   455→                          <Checkbox
   456→   456→                            checked={selectedIds.has(product.id)}
   457→   457→                            onCheckedChange={() => toggleSelect(product.id)}
   458→   458→                          />
   459→   459→                        </TableCell>
   460→   460→                        <TableCell>
   461→   461→                          <OptimizedImage
   462→   462→                            src={product.images?.[0]?.url || '/logo-kabyle.png'}
   463→   463→                            alt={product.name}
   464→   464→                            width={40}
   465→   465→                            height={40}
   466→   466→                            className="rounded-md"
   467→   467→                            useThumbnail
   468→   468→                          />
   469→   469→                        </TableCell>
   470→   470→                        <TableCell className="font-mono text-xs">{product.reference}</TableCell>
   471→   471→                        <TableCell className="font-medium max-w-[200px] truncate">{product.name}</TableCell>
   472→   472→                        <TableCell className="text-sm">{formatPrice(product.price)}</TableCell>
   473→   473→                        <TableCell>
   474→   474→                          <span
   475→   475→                            className={`text-sm font-medium ${
   476→   476→                              getTotalStock(product) === 0
   477→   477→                                ? 'text-red-600'
   478→   478→                                : getTotalStock(product) <= 3
   479→   479→                                ? 'text-yellow-600'
   480→   480→                                : 'text-foreground'
   481→   481→                            }`}
   482→   482→                          >
   483→   483→                            {getTotalStock(product)}
   484→   484→                          </span>
   485→   485→                        </TableCell>
   486→   486→                        <TableCell>
   487→   487→                          <div className="flex gap-1">
   488→   488→                            {product.active ? (
   489→   489→                              <Badge variant="outline" className="bg-green-100 text-green-800 border-green-200">
   490→   490→                                Actif
   491→   491→                              </Badge>
   492→   492→                            ) : (
