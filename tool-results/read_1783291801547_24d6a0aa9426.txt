    89→    89→
    90→    90→export default function ProductsPage() {
    91→    91→  const router = useRouter();
    92→    92→  const [products, setProducts] = useState<Product[]>([]);
    93→    93→  const [pagination, setPagination] = useState<Pagination>({ page: 1, limit: 20, total: 0, totalPages: 0 });
    94→    94→  const [loading, setLoading] = useState(true);
    95→    95→  const [search, setSearch] = useState('');
    96→    96→
    97→    97→  // Selection state
    98→    98→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
    99→    99→
   100→   100→  // Dialog states
   101→   101→  const [dialogOpen, setDialogOpen] = useState(false);
   102→   102→  const [editProduct, setEditProduct] = useState<Product | null>(null);
   103→   103→  const [saving, setSaving] = useState(false);
   104→   104→
   105→   105→  // Delete state (single)
   106→   106→  const [deleteProduct, setDeleteProduct] = useState<Product | null>(null);
   107→   107→  const [deleting, setDeleting] = useState(false);
   108→   108→
   109→   109→  // Bulk delete state
   110→   110→  const [bulkDeleteOpen, setBulkDeleteOpen] = useState(false);
   111→   111→  const [bulkDeleting, setBulkDeleting] = useState(false);
   112→   112→
   113→   113→  // Form state
   114→   114→  const [formReference, setFormReference] = useState('');
   115→   115→  const [formName, setFormName] = useState('');
   116→   116→  const [formDescription, setFormDescription] = useState('');
   117→   117→  const [formPrice, setFormPrice] = useState('');
   118→   118→  const [formFabric, setFormFabric] = useState('');
   119→   119→  const [formFeatured, setFormFeatured] = useState(false);
   120→   120→  const [formActive, setFormActive] = useState(true);
   121→   121→  const [formVariants, setFormVariants] = useState<Array<{ size: string; color: string; stock: string }>>([
   122→   122→    { size: '', color: '', stock: '0' },
   123→   123→  ]);
   124→   124→  const [formImages, setFormImages] = useState<Array<{ url: string; alt?: string; sortOrder: number }>>([]);
   125→   125→  const [imageUrlInput, setImageUrlInput] = useState('');
   126→   126→
   127→   127→  const fetchProducts = useCallback(async (page = 1) => {
   128→   128→    try {
   129→   129→      setLoading(true);
   130→   130→      const params = new URLSearchParams();
   131→   131→      if (search) params.set('search', search);
   132→   132→      params.set('page', page.toString());
   133→   133→      params.set('limit', '20');
   134→   134→
   135→   135→      const res = await fetch(`/api/products?${params}`);
   136→   136→      if (res.ok) {
   137→   137→        const data = await res.json();
   138→   138→        setProducts(data.products);
   139→   139→        setPagination(data.pagination);
   140→   140→        setSelectedIds(new Set()); // Clear selection on new data
   141→   141→      }
   142→   142→    } catch (error) {
   143→   143→      console.error('Error fetching products:', error);
   144→   144→    } finally {
   145→   145→      setLoading(false);
   146→   146→    }
   147→   147→  }, [search]);
   148→   148→
   149→   149→  useEffect(() => {
   150→   150→    fetchProducts(1);
   151→   151→  }, [fetchProducts]);
   152→   152→
   153→   153→  // Selection helpers
   154→   154→  const toggleSelect = (id: string) => {
   155→   155→    setSelectedIds((prev) => {
   156→   156→      const next = new Set(prev);
   157→   157→      if (next.has(id)) {
   158→   158→        next.delete(id);
   159→   159→      } else {
   160→   160→        next.add(id);
   161→   161→      }
   162→   162→      return next;
   163→   163→    });
   164→   164→  };
   165→   165→
   166→   166→  const toggleSelectAll = () => {
   167→   167→    if (selectedIds.size === products.length) {
   168→   168→      setSelectedIds(new Set());
   169→   169→    } else {
   170→   170→      setSelectedIds(new Set(products.map((p) => p.id)));
   171→   171→    }
   172→   172→  };
   173→   173→
   174→   174→  const allSelected = products.length > 0 && selectedIds.size === products.length;
   175→   175→  const someSelected = selectedIds.size > 0 && !allSelected;
   176→   176→
   177→   177→  const openCreateDialog = () => {
   178→   178→    setEditProduct(null);
   179→   179→    setFormReference('');
   180→   180→    setFormName('');
   181→   181→    setFormDescription('');
   182→   182→    setFormPrice('');
   183→   183→    setFormFabric('');
   184→   184→    setFormFeatured(false);
   185→   185→    setFormActive(true);
   186→   186→    setFormVariants([{ size: '', color: '', stock: '0' }]);
   187→   187→    setFormImages([]);
   188→   188→    setDialogOpen(true);
   189→   189→  };
   190→   190→
   191→   191→  const openEditDialog = (product: Product) => {
   192→   192→    setEditProduct(product);
   193→   193→    setFormReference(product.reference);
   194→   194→    setFormName(product.name);
   195→   195→    setFormDescription(product.description || '');
   196→   196→    setFormPrice(product.price.toString());
   197→   197→    setFormFabric(product.fabric || '');
   198→   198→    setFormFeatured(product.featured);
   199→   199→    setFormActive(product.active);
   200→   200→    setFormVariants(
   201→   201→      product.variants.length > 0
   202→   202→        ? product.variants.map((v) => ({ size: v.size, color: v.color, stock: v.stock.toString() }))
   203→   203→        : [{ size: '', color: '', stock: '0' }]
   204→   204→    );
   205→   205→    setFormImages(
   206→   206→      product.images.map((img, idx) => ({ url: img.url, alt: img.alt || undefined, sortOrder: idx }))
   207→   207→    );
   208→   208→    setDialogOpen(true);
   209→   209→  };
   210→   210→
   211→   211→  const handleSave = async () => {
   212→   212→    if (!formReference || !formName || !formPrice) {
   213→   213→      toast.error('Veuillez remplir la référence, le nom et le prix');
   214→   214→      return;
   215→   215→    }
   216→   216→
   217→   217→    const validVariants = formVariants.filter((v) => v.size && v.color);
   218→   218→
   219→   219→    const payload = {
   220→   220→      reference: formReference,
   221→   221→      name: formName,
   222→   222→      description: formDescription || null,
   223→   223→      price: parseFloat(formPrice),
   224→   224→      fabric: formFabric || null,
   225→   225→      featured: formFeatured,
   226→   226→      active: formActive,
   227→   227→      variants: validVariants.map((v) => ({ size: v.size, color: v.color, stock: parseInt(v.stock) || 0 })),
   228→   228→      images: formImages,
   229→   229→    };
   230→   230→
   231→   231→    try {
   232→   232→      setSaving(true);
   233→   233→      let res;
   234→   234→      if (editProduct) {
   235→   235→        res = await fetch(`/api/products/${editProduct.id}`, {
   236→   236→          method: 'PUT',
   237→   237→          headers: { 'Content-Type': 'application/json' },
   238→   238→          body: JSON.stringify(payload),
   239→   239→        });
   240→   240→      } else {
   241→   241→        res = await fetch('/api/products', {
   242→   242→          method: 'POST',
   243→   243→          headers: { 'Content-Type': 'application/json' },
   244→   244→          body: JSON.stringify(payload),
   245→   245→        });
   246→   246→      }
   247→   247→
   248→   248→      if (res.ok) {
   249→   249→        toast.success(editProduct ? 'Produit mis à jour avec succès' : 'Produit créé avec succès');
   250→   250→        setDialogOpen(false);
   251→   251→        fetchProducts(pagination.page);
   252→   252→      } else {
   253→   253→        const data = await res.json();
   254→   254→        toast.error(data.error || 'Erreur lors de la sauvegarde');
   255→   255→      }
   256→   256→    } catch {
   257→   257→      toast.error('Erreur lors de la sauvegarde');
   258→   258→    } finally {
   259→   259→      setSaving(false);
   260→   260→    }
   261→   261→  };
   262→   262→
   263→   263→  const handleDelete = async () => {
   264→   264→    if (!deleteProduct) return;
   265→   265→
   266→   266→    try {
   267→   267→      setDeleting(true);
   268→   268→      const res = await fetch(`/api/products/${deleteProduct.id}`, { method: 'DELETE' });
   269→   269→      if (res.ok) {
   270→   270→        toast.success('Produit supprimé avec succès');
   271→   271→        setDeleteProduct(null);
   272→   272→        fetchProducts(pagination.page);
   273→   273→      } else {
   274→   274→        const data = await res.json();
   275→   275→        toast.error(data.error || 'Erreur lors de la suppression');
   276→   276→      }
   277→   277→    } catch {
   278→   278→      toast.error('Erreur lors de la suppression');
   279→   279→    } finally {
   280→   280→      setDeleting(false);
   281→   281→    }
   282→   282→  };
   283→   283→
   284→   284→  const handleBulkDelete = async () => {
   285→   285→    if (selectedIds.size === 0) return;
   286→   286→
   287→   287→    try {
   288→   288→      setBulkDeleting(true);
   289→   289→      const res = await fetch('/api/products', {
   290→   290→        method: 'DELETE',
   291→   291→        headers: { 'Content-Type': 'application/json' },
   292→   292→        body: JSON.stringify({ ids: Array.from(selectedIds) }),
   293→   293→      });
   294→   294→      if (res.ok) {
   295→   295→        const data = await res.json();
   296→   296→        toast.success(data.message || `${selectedIds.size} produits supprimés avec succès`);
   297→   297→        setSelectedIds(new Set());
   298→   298→        setBulkDeleteOpen(false);
   299→   299→        fetchProducts(pagination.page);
   300→   300→      } else {
   301→   301→        const data = await res.json();
   302→   302→        toast.error(data.error || 'Erreur lors de la suppression');
   303→   303→      }
   304→   304→    } catch {
   305→   305→      toast.error('Erreur lors de la suppression multiple');
   306→   306→    } finally {
   307→   307→      setBulkDeleting(false);
   308→   308→    }
   309→   309→  };
   310→   310→
   311→   311→  const addImageUrl = () => {
   312→   312→    const url = imageUrlInput.trim();
   313→   313→    if (!url) {
   314→   314→      toast.error('Veuillez entrer une URL d\'image');
   315→   315→      return;
   316→   316→    }
   317→   317→    // Basic URL validation
   318→   318→    try {
   319→   319→      new URL(url);
   320→   320→    } catch {
   321→   321→      toast.error('URL invalide. Veuillez entrer une URL valide (ex: https://...)');
   322→   322→      return;
   323→   323→    }
   324→   324→    setFormImages((prev) => [...prev, { url, sortOrder: prev.length }]);
   325→   325→    setImageUrlInput('');
   326→   326→  };
   327→   327→
   328→   328→  const removeImage = (index: number) => {
   329→   329→    setFormImages((prev) => prev.filter((_, i) => i !== index).map((img, i) => ({ ...img, sortOrder: i })));
   330→   330→  };
   331→   331→
   332→   332→  const addVariant = () => {
   333→   333→    setFormVariants((prev) => [...prev, { size: '', color: '', stock: '0' }]);
   334→   334→  };
   335→   335→
   336→   336→  const removeVariant = (index: number) => {
   337→   337→    setFormVariants((prev) => prev.filter((_, i) => i !== index));
   338→   338→  };
   339→   339→
   340→   340→  const updateVariant = (index: number, field: 'size' | 'color' | 'stock', value: string) => {
   341→   341→    setFormVariants((prev) =>
   342→   342→      prev.map((v, i) => (i === index ? { ...v, [field]: value } : v))
   343→   343→    );
   344→   344→  };
   345→   345→
   346→   346→  const getTotalStock = (product: Product) => {
   347→   347→    return product.variants.reduce((sum, v) => sum + v.stock, 0);
   348→   348→  };
   349→   349→
   350→   350→  return (
   351→   351→    <div className="space-y-6">
   352→   352→      {/* Header */}
   353→   353→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   354→   354→        <div>
   355→   355→          <h2 className="text-2xl font-bold">Produits</h2>
   356→   356→          <p className="text-muted-foreground">{pagination.total} produits au total</p>
   357→   357→        </div>
   358→   358→        <div className="flex items-center gap-2">
   359→   359→          {selectedIds.size > 0 && (
   360→   360→            <Button
   361→   361→              variant="destructive"
   362→   362→              onClick={() => setBulkDeleteOpen(true)}
   363→   363→              className="shrink-0"
   364→   364→            >
   365→   365→              <Trash2 className="mr-2 h-4 w-4" />
   366→   366→              Supprimer ({selectedIds.size})
   367→   367→            </Button>
   368→   368→          )}
   369→   369→          <Button onClick={openCreateDialog} className="bg-primary hover:bg-primary/90">
   370→   370→            <Plus className="mr-2 h-4 w-4" />
   371→   371→            Ajouter un produit
   372→   372→          </Button>
   373→   373→        </div>
   374→   374→      </div>
   375→   375→
   376→   376→      {/* Search */}
   377→   377→      <div className="relative max-w-md">
   378→   378→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   379→   379→        <Input
   380→   380→          placeholder="Rechercher par référence, nom..."
   381→   381→          value={search}
   382→   382→          onChange={(e) => setSearch(e.target.value)}
   383→   383→          className="pl-10"
   384→   384→        />
   385→   385→      </div>
   386→   386→
   387→   387→      {/* Selection info bar */}
   388→   388→      {selectedIds.size > 0 && (
   389→   389→        <div className="flex items-center gap-3 px-4 py-2 bg-muted/50 rounded-lg border">
   390→   390→          <span className="text-sm font-medium">
   391→   391→            {selectedIds.size} produit{selectedIds.size > 1 ? 's' : ''} sélectionné{selectedIds.size > 1 ? 's' : ''}
   392→   392→          </span>
   393→   393→          <Button
   394→   394→            variant="ghost"
   395→   395→            size="sm"
   396→   396→            className="text-xs h-7"
   397→   397→            onClick={() => setSelectedIds(new Set())}
   398→   398→          >
   399→   399→            Tout désélectionner
   400→   400→          </Button>
   401→   401→        </div>
   402→   402→      )}
   403→   403→
   404→   404→      {/* Products Table */}
   405→   405→      <Card>
   406→   406→        <CardContent className="p-0">
   407→   407→          {loading ? (
   408→   408→            <div className="p-6 space-y-4">
   409→   409→              {Array.from({ length: 5 }).map((_, i) => (
   410→   410→                <Skeleton key={i} className="h-16 w-full" />
   411→   411→              ))}
   412→   412→            </div>
   413→   413→          ) : products.length === 0 ? (
   414→   414→            <div className="flex flex-col items-center justify-center py-16 text-muted-foreground">
   415→   415→              <ImageIcon className="h-12 w-12 mb-4 opacity-50" />
   416→   416→              <p className="text-lg font-medium">Aucun produit trouvé</p>
   417→   417→              <p className="text-sm">Commencez par ajouter votre premier produit</p>
   418→   418→            </div>
   419→   419→          ) : (
   420→   420→            <>
   421→   421→              <div className="overflow-x-auto">
   422→   422→                <Table>
   423→   423→                  <TableHeader>
   424→   424→                    <TableRow>
   425→   425→                      <TableHead className="w-12">
   426→   426→                        <Checkbox
   427→   427→                          checked={allSelected}
   428→   428→                          ref={(el) => {
   429→   429→                            if (el) {
   430→   430→                              (el as HTMLButtonElement & { indeterminate?: boolean }).indeterminate = someSelected;
   431→   431→                            }
   432→   432→                          }}
   433→   433→                          onCheckedChange={toggleSelectAll}
   434→   434→                        />
   435→   435→                      </TableHead>
   436→   436→                      <TableHead className="w-16">Image</TableHead>
   437→   437→                      <TableHead>Référence</TableHead>
   438→   438→                      <TableHead>Nom</TableHead>
   439→   439→                      <TableHead>Prix</TableHead>
   440→   440→                      <TableHead>Stock total</TableHead>
   441→   441→                      <TableHead>Statut</TableHead>
   442→   442→                      <TableHead className="text-right">Actions</TableHead>
   443→   443→                    </TableRow>
   444→   444→                  </TableHeader>
   445→   445→                  <TableBody>
   446→   446→                    {products.map((product) => (
   447→   447→                      <TableRow
   448→   448→                        key={product.id}
   449→   449→                        className={selectedIds.has(product.id) ? 'bg-muted/50' : ''}
   450→   450→                      >
   451→   451→                        <TableCell>
   452→   452→                          <Checkbox
   453→   453→                            checked={selectedIds.has(product.id)}
   454→   454→                            onCheckedChange={() => toggleSelect(product.id)}
   455→   455→                          />
   456→   456→                        </TableCell>
   457→   457→                        <TableCell>
   458→   458→                          {product.images[0] ? (
   459→   459→                            <img
   460→   460→                              src={product.images[0].url}
   461→   461→                              alt={product.name}
   462→   462→                              className="w-10 h-10 rounded-md object-cover"
   463→   463→                            />
   464→   464→                          ) : (
   465→   465→                            <div className="w-10 h-10 rounded-md bg-muted flex items-center justify-center">
   466→   466→                              <ImageIcon className="h-4 w-4 text-muted-foreground" />
   467→   467→                            </div>
   468→   468→                          )}
   469→   469→                        </TableCell>
   470→   470→                        <TableCell className="font-mono text-xs">{product.reference}</TableCell>
   471→   471→                        <TableCell className="font-medium max-w-[200px] truncate">{product.name}</TableCell>
   472→   472→                        <TableCell className="text-sm">{formatPrice(product.price)}</TableCell>
   473→   473→                        <TableCell>
   474→   474→                          <span
   475→   475→                            className={`text-sm font-medium ${
   476→   476→                              getTotalStock(product) === 0
   477→   477→                                ? 'text-red-600'
   478→   478→                                : getTotalStock(product) <= 3
   479→   479→                                ? 'text-yellow-600'
   480→   480→                                : 'text-foreground'
   481→   481→                            }`}
   482→   482→                          >
   483→   483→                            {getTotalStock(product)}
   484→   484→                          </span>
   485→   485→                        </TableCell>
   486→   486→                        <TableCell>
   487→   487→                          <div className="flex gap-1">
   488→   488→                            {product.active ? (
   489→   489→                              <Badge variant="outline" className="bg-green-100 text-green-800 border-green-200">
   490→   490→                                Actif
   491→   491→                              </Badge>
